void glfwInputPolling(GLFWwindow* window, player &player1)
{
    if(glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS)
        player1.position.y -= 0.01;
    if(glfwGetKey(window, GLFW_KEY_S) == GLFW_PRESS)
        player1.position.y += 0.01;
    if(glfwGetKey(window, GLFW_KEY_D) == GLFW_PRESS)
        player1.position.x -= 0.01;
    if(glfwGetKey(window, GLFW_KEY_A) == GLFW_PRESS)
        player1.position.x += 0.01;
    if(glfwGetKey(window, GLFW_KEY_SPACE) == GLFW_PRESS)
        std::cout << "";
    if(glfwGetKey(window, GLFW_KEY_CAPS_LOCK) == GLFW_PRESS)
        std::cout << "";
}

void createGrid(VAO &vao, VBO &vbo, EBO &ebo, bool &gridLoaded, int gridSize, float gridBoxLength)
{   
    /*vertices array size ----> 
    because
    grid of size 1 
    requires 4 points
    which makes (gridsize + 1)^2
    1+1*1+1 = 4

    then multiply with 8 cuz every vertex requires 8 properties
    */
    GLfloat 
    r = 1,
    g = 1,
    b = 1,
    u = 0,
    v = 0;

    GLfloat yPos = 0 - (((float)sqrt(gridSize)) * gridBoxLength);
    GLfloat xPos = 0 - (((float)sqrt(gridSize)) * gridBoxLength);
    GLint verticesArraySize = ((gridSize + 1)*(gridSize + 1)) * 8;
    GLint verticesArrayLength = (gridSize + 1) * 8; 
    GLfloat *vertices = new GLfloat[verticesArraySize];  
    int currentIndex = 0;
    std::cout
    << "\nstartingX:" << xPos 
    << "\nstartingY:" << yPos
    << "\nverticesLength:" << verticesArrayLength
    << "\nverticesSize:" << verticesArraySize
    << "\n";
    for(int y = 0; y <= gridSize; y++)
    {
        for(int x = 0; x <= gridSize; x++)
        {
            std::cout << "X" << xPos << "," << yPos << "Y" << "|";
            vertices[currentIndex+0] = xPos;
            vertices[currentIndex+1] = 0;
            vertices[currentIndex+2] = yPos;
            vertices[currentIndex+3] = r;
            vertices[currentIndex+4] = g;
            vertices[currentIndex+5] = b;
            vertices[currentIndex+6] = u;
            vertices[currentIndex+7] = v;
            xPos += gridBoxLength;
            currentIndex += 8;
        }
        xPos = 0 - (((float)sqrt(gridSize)) * gridBoxLength);
        yPos += gridBoxLength;
    }

    //dumpppppp
    
    std::cout << "\n\nvertices info dump:\n";

    int xs = 1;
    for (int i = 0; i < verticesArraySize; i++) 
    {
        
        if(xs == 9)
        {
            xs = 1;
            std::cout << "------";
        }
        xs++;
        std::cout << vertices[i] << ",";
        
    };

    //indices

    GLint indicesAmount = (2*gridSize*(gridSize+1)) * 2;
    GLuint *indices = new GLuint[indicesAmount];
    GLuint indicesPerRow = gridSize;

    std::cout
    << "\nIndices Amount:" << indicesAmount
    << "\nindicesPerRow:" << indicesPerRow
    << "\n";

    int currentLineIndex = 1;
    int currentIndicesYIndex = 0;
    int yAddition = 0;
    for(int x = 0; x <= indicesPerRow; x++)
    {
        for(int y = 0; y < indicesPerRow; y++)
        {
            indices[currentLineIndex-1] = y+currentIndicesYIndex+yAddition;
            std::cout << "\ncurrentLineIndex:" << currentLineIndex;
            std::cout << "\nwith value:" << y+currentIndicesYIndex+yAddition;
            std::cout << "\ncurrentIndicesYindez";
            indices[currentLineIndex] = y+1+currentIndicesYIndex+yAddition;
            currentLineIndex+=2;
        }
        yAddition++;
        currentIndicesYIndex += indicesPerRow;
    }
    int addY = 0;
    int addX = gridSize+1;
    for(int x = 0; x <= gridSize; x++)
    {
        for(int y = 0; y < gridSize; y++)
        {
            indices[currentLineIndex-1] = addX * y + addY;
            indices[currentLineIndex] = addX * (y+1) + addY;
            currentLineIndex+=2;
        }
        addY++;
    }


    std::cout << "\n" << (float)sizeof(indices);
    //dumpppppp
    std::cout << "\n\nIndices dump:\n\n";
    int xf = 1;
    for (int i = 0; i < indicesAmount; i++) 
    {
        
        if(xf == 3)
        {
            xf = 1;
            std::cout << "------";
        }
        xf++;
        std::cout << indices[i] << ",";
        
    };

// 1,2 2,3 3,4
// 0,1,2,3,4,5
    vao.bind();
    vbo.setData(vertices, verticesArraySize*sizeof(GLfloat));
    ebo.setData(indices, indicesAmount*sizeof(GLuint));
    vao.linkAttribute(vbo ,0, 3, GL_FLOAT, 8 * sizeof(float), (void*)0);
    vao.linkAttribute(vbo ,1, 3, GL_FLOAT, 8 * sizeof(float), (void*)(3*sizeof(float)));
    vao.linkAttribute(vbo ,2, 2, GL_FLOAT, 8 * sizeof(float), (void*)(6*sizeof(float)));
    gridLoaded = true;
    delete[] indices;
    delete[] vertices;
}
